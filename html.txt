    <script>
        // Constants / Global Variables
        let currentDiet = 'optimized'; // or 'standard'
        let chartInstance = null;

        // Model Constants:
        // W(t) = a*t^3 + b*t^2 + c*t + d
        // W'(t) = 3*a*t^2 + 2*b*t + c
        const models = {
            optimized: {
                a: -0.015,
                b: 0.65,
                c: 5.5,
                d: 15,
                name: "Intensive Feedlot Diet (PKC + Napier)",
                formulaTex: "W(t) = -0.015t^3 + 0.65t^2 + 5.5t + 15",
                derivTex: "W'(t) = -0.045t^2 + 1.30t + 5.50"
            },
            standard: {
                a: -0.012,
                b: 0.52,
                c: 4.5,
                d: 14,
                name: "Standard Extensive Diet (Napier Only)",
                formulaTex: "W(t) = -0.012t^3 + 0.52t^2 + 4.5t + 14",
                derivTex: "W'(t) = -0.036t^2 + 1.04t + 4.50"
            }
        };

        // DOM elements
        const ageSlider = document.getElementById('ageSlider');
        const sliderVal = document.getElementById('slider-val');
        const badgeAge = document.getElementById('current-age-badge');
        const calcAge = document.getElementById('calc-age');
        const calcDays = document.getElementById('calc-days');
        const calcWeight = document.getElementById('calc-weight');
        const calcWeightEq = document.getElementById('calc-weight-equation');
        const calcDerivative = document.getElementById('calc-derivative');
        const calcDailyRate = document.getElementById('calc-daily-rate');
        const laymanHeadline = document.getElementById('layman-headline');
        const laymanText = document.getElementById('layman-text');
        
        const dietOptBtn = document.getElementById('diet-opt-btn');
        const dietStdBtn = document.getElementById('diet-std-btn');

        // Main calculations based on selected diet and age
        function calculateValues(t, diet) {
            const m = models[diet];
            
            // Weight
            const w = (m.a * Math.pow(t, 3)) + (m.b * Math.pow(t, 2)) + (m.c * t) + m.d;
            
            // Derivative (instantaneous growth rate per month)
            // W'(t) = 3*a*t^2 + 2*b*t + c
            const wPrime = (3 * m.a * Math.pow(t, 2)) + (2 * m.b * t) + m.c;
            
            // Daily rate estimation: Monthly rate divided by 30.4 days
            const dailyGrams = Math.round((wPrime / 30.41) * 1000);

            return {
                weight: Math.max(0, w).toFixed(2),
                weightRaw: w,
                derivative: wPrime.toFixed(2),
                derivativeRaw: wPrime,
                dailyGrams: dailyGrams
            };
        }

        // Get custom layman narrative based on cattle lifecycle stages
        function getLaymanExplanation(t, diet, values) {
            let headline = "";
            let text = "";

            if (t <= 6) { // Pre-weaning / Early growth phase
                headline = "Early Developmental Phase (Suckling to Early Weaning)";
                if (diet === 'optimized') {
                    text = `At ${t} months, the calf is highly efficient at processing milk protein alongside Napier grass. Weight gain derivative is high (${values.derivative} kg/month). Since protein retention rates are peaking, feeding standard mineral supplement blocks in addition to pasture keeps development on its optimal trajectory.`;
                } else {
                    text = `At ${t} months, under extensive grazing, growth is sound but lacks development speed (${values.derivative} kg/month). Supplementation with concentrate feeds would significantly accelerate this slope by improving rumen development.`;
                }
            } 
            else if (t > 6 && t <= 18) { // Puberty & Peak physiological acceleration
                // Max growth acceleration is at inflection point
                const inflectionPoint = diet === 'optimized' ? 14.4 : 14.4; 
                
                if (Math.abs(t - inflectionPoint) <= 2) {
                    headline = "🚨 Growth Inflection Point: Peak Feed Efficiency!";
                    text = `At ${t} months, your cattle has reached its highest possible growth velocity (W' is maximized!). This is the optimal window to push feed allocation. Rumen microbes are converting bypass proteins like Palm Kernel Cake with maximum efficiency into skeletal structure. FCR is at its historical best. Keep feed intake high to maximize profits!`;
                } else {
                    headline = "Post-Weaning Lean Muscle Growth Stage";
                    text = `At ${t} months, the animal is putting on rapid muscular tissue. The derivative shows a sustained high rate of ${values.derivative} kg/month (${values.dailyGrams} g/day). Feeding nutrient-dense proteins guarantees excellent beef marbling and frame development prior to the onset of early maturity.`;
                }
            } 
            else if (t > 18 && t <= 28) { // Maturity onset / decelerating growth
                headline = "Maturity Onset: Decreasing Growth Acceleration";
                text = `At ${t} months, growth rate (derivative) has dropped to ${values.derivative} kg/month. The cattle is reaching adult weight. The biological system shifts from muscle deposition to fat storage. Note that the derivative is still positive ($f'(x) > 0$), meaning the animal continues to gain weight, but at a decelerating pace. Feed economics must be checked carefully!`;
            } 
            else { // Terminal finishing phase / Plateau
                headline = "Market Maturity Plateau (Optimal Slaughter Point)";
                if (values.derivativeRaw < 2.0) {
                    text = `At ${t} months, the growth rate has plummeted to a negligible ${values.derivative} kg/month (around ${values.dailyGrams}g/day). Because the weight derivative is close to zero, the steer is consuming feed purely for standard body maintenance. From an animal husbandry and business perspective, keeping the animal past this stage results in massive financial losses on feeding costs. Send to market immediately!`;
                } else {
                    text = `At ${t} months, growth is plateauing. Daily gain is low. Monitor weight changes: as $W'(t)$ approaches zero, feed efficiency is very poor. Plan herd turnover and scheduling of slaughtering facilities to capitalize on peak pricing.`;
                }
            }

            return { headline, text };
        }

        // Generate data arrays for Chart.js
        function getChartDatasets(diet) {
            const labels = [];
            const weightData = [];
            const rateData = [];

            for (let t = 0; t <= 36; t += 1) {
                labels.push(t);
                const val = calculateValues(t, diet);
                weightData.push(val.weightRaw);
                rateData.push(val.derivativeRaw);
            }

            return { labels, weightData, rateData };
        }

        }

        // Update calculations panel
        function updateCalculationsPanel(t) {
            const vals = calculateValues(t, currentDiet);
            
            // Slider outputs
            sliderVal.innerText = t;
            badgeAge.innerText = t + " Months";
            calcAge.innerHTML = `${t} <span class="text-sm font-normal text-slate-400">months</span>`;
            
            // Convert months to approx days
            const days = Math.round(t * 30.41);
            calcDays.innerText = days;

            calcWeight.innerHTML = `${vals.weight} <span class="text-sm font-normal text-slate-400">kg</span>`;
            
            // Generate standard equation presentation
            const m = models[currentDiet];
            calcWeightEq.innerText = `W(${t}) = ${m.a}(${t})³ + ${m.b}(${t})² + ${m.c}(${t}) + ${m.d}`;

            calcDerivative.innerHTML = `${vals.derivative} <span class="text-sm font-normal text-emerald-500/80">kg/month</span>`;
            calcDailyRate.innerText = `${vals.dailyGrams} g/day`;

            // Narrative Explainer
            const exp = getLaymanExplanation(t, currentDiet, vals);
            laymanHeadline.innerText = exp.headline;
            laymanText.innerText = exp.text;
        }

            const parsedVal = parseFloat(val);
            updateCalculationsPanel(parsedVal);

        function switchDiet(diet) {
            currentDiet = diet;
            
            // Visual toggle behavior
            if (diet === 'optimized') {
                dietOptBtn.className = "w-full text-left p-4 rounded-xl border transition-all duration-200 bg-emerald-950/30 border-emerald-500/50 text-white shadow-md shadow-emerald-500/5";
                dietStdBtn.className = "w-full text-left p-4 rounded-xl border transition-all duration-200 bg-slate-950 border-slate-800 text-slate-400 hover:border-slate-700 hover:text-white";
            } else {
                dietOptBtn.className = "w-full text-left p-4 rounded-xl border transition-all duration-200 bg-slate-950 border-slate-800 text-slate-400 hover:border-slate-700 hover:text-white";
                dietStdBtn.className = "w-full text-left p-4 rounded-xl border transition-all duration-200 bg-emerald-950/30 border-emerald-500/50 text-white shadow-md shadow-emerald-500/5";
            }

            // Formulas Swap block
            const formMath = document.getElementById('model-formula-math');
            const derivMath = document.getElementById('derivative-formula-math');
            
            if (diet === 'optimized') {
                formMath.innerText = '$$W(t) = -0.015t^3 + 0.65t^2 + 5.5t + 15$$';
                derivMath.innerText = `$$W'(t) = -0.045t^2 + 1.30t + 5.50$$`;
            } else {
                formMath.innerText = '$$W(t) = -0.012t^3 + 0.52t^2 + 4.5t + 14$$';
                derivMath.innerText = `$$W'(t) = -0.036t^2 + 1.04t + 4.50$$`;
            }

            // Re-render MathJax dynamic math formulations
            if (window.MathJax && window.MathJax.typesetPromise) {
                window.MathJax.typesetPromise();
            }

            // Refresh visuals
            updateChart(diet);
            handleSliderChange(ageSlider.value);
        }
